-- the wheelbarrow at the tree pump (tr): pushed (fwd) or pulled back, as the keys do, until stop() or the end
guide_barrow_move = function(fwd, stop)
  if fwd then
    tr.set_walk_backwards(FALSE)
    manny:play_chore_looping(ma_use_wb_walk, "ma_use_wb.cos")
    tr.barrow:play_chore_looping(wheelbarrow_ma_walk, "wheelbarrow.cos")
    while not stop() and not manny:find_sector_name("wb_push_lim") do
      manny:walk_forward()
      tr.barrow:walk_forward()
      break_here()
    end
    manny:stop_chore(ma_use_wb_walk, "ma_use_wb.cos")
  else
    tr.set_walk_backwards(TRUE)
    manny:stop_chore()
    manny:play_chore_looping(ma_use_wb_walk_back, "ma_use_wb.cos")
    tr.barrow:play_chore_looping(wheelbarrow_ma_walk, "wheelbarrow.cos")
    while not stop() and not manny:find_sector_name("wb_pull_lim") do
      manny:walk_forward()
      tr.barrow:walk_forward()
      break_here()
    end
    manny:stop_chore(ma_use_wb_walk_back, "ma_use_wb.cos")
    tr.set_walk_backwards(FALSE)
  end
  tr.barrow:stop_chore(wheelbarrow_ma_walk, "wheelbarrow.cos")
  manny:play_chore(ma_use_wb_hold_wb, "ma_use_wb.cos")
  tr.barrow:play_chore(wheelbarrow_ma_hold, "wheelbarrow.cos")
end
guide_barrow_box = function()
  local id, name = tr.barrow:find_sector_type(HOT)
  return name
end
-- waits for the pumps' next beat that passes ok(beat)
guide_wait_beat = function(ok)
  local b = tr.beat
  while TRUE do
    while tr.beat == b do break_here() end
    b = tr.beat
    if ok(b) then return end
  end
end
-- onto a hose (moving fwd or back), until the right beat, then off it the way it came: the hose's pump then
-- beats on that beat (tr.hose_watch)
guide_hose = function(box, fwd, ok)
  guide_barrow_move(fwd, function() return guide_barrow_box() == %box end)
  guide_wait_beat(ok)
  guide_barrow_move(not fwd, function() return guide_barrow_box() == nil end)
end
-- the tree rocks when the two pumps on each side beat together, and the sides don't (tr.pump_timer_and_checker).
-- Along the barrow's path: hose_box4 (pump1), hose_box3 (pump2), hose_box2 (pump3), hose_box1 (pump4).
guide_tree_pumps = function()
  local n = 0
  -- the right side first: pump3 onto pump4's beat, coming from the left
  repeat
    guide_hose("hose_box2", tr.barrow:getpos().x < 3.3, function(b) return b == tr.pump4.beat end)
    n = n + 1
  until tr.pump3.beat == tr.pump4.beat or n > 6
  -- the left side must beat apart from the right: pump1 over again if not (never crossing the right hoses)
  n = 0
  while tr.pump1.beat == tr.pump3.beat and n < 6 do
    guide_hose("hose_box4", FALSE, function(b) return b ~= tr.pump3.beat end)
    n = n + 1
  end
  -- then pump2 onto pump1's beat
  n = 0
  repeat
    guide_hose("hose_box3", tr.barrow:getpos().x < 3.05, function(b) return b == tr.pump1.beat end)
    n = n + 1
  until tr.pump2.beat == tr.pump1.beat or n > 6
end

-- the vault's combination (vd): four tumblers driven by the wheel; each picks up the next one when it reaches
-- that one's pin, and turning the other way lets 2-4 go (Safe.monitor_lock).  A plan is found on a copy of
-- them, one degree at a time: clockwise until the fourth sits in the open window (248-290), back until the
-- third does without the fourth moving, and so on; then the wheel is turned that way.
guide_vault_copy = function(t)
  local s = { r = { }, l = { }, v = { }, p = { } }
  local i = 1
  while i <= 4 do
    s.r[i] = t[i].rot
    s.l[i] = t[i].lock
    s.v[i] = t[i].lock_val
    s.p[i] = t[i].lock_pin
    i = i + 1
  end
  return s
end
guide_vault_clone = function(s)
  local c = { r = { }, l = { }, v = { }, p = { } }
  local i = 1
  while i <= 4 do
    c.r[i] = s.r[i]
    c.l[i] = s.l[i]
    c.v[i] = s.v[i]
    c.p[i] = s.p[i]
    i = i + 1
  end
  return c
end
-- a turn the other way: tumblers 2-4 let go, each to be picked up again lock_pin degrees on
guide_vault_reverse = function(s)
  local i = 2
  while i <= 4 do
    s.l[i] = FALSE
    s.v[i] = s.r[i] + s.p[i]
    if s.v[i] >= 360 then s.v[i] = s.v[i] - 360 end
    i = i + 1
  end
end
-- one degree (d = 1 clockwise, -1 back), as Safe.spin_tumblers_right/left
guide_vault_step = function(s, d)
  local i = 1
  while i <= 4 do
    if s.l[i] then
      s.r[i] = s.r[i] + d
      if s.r[i] == 360 then s.r[i] = 0 end
      if s.r[i] == -1 then s.r[i] = 359 end
      if i < 4 and s.r[i] == s.v[i + 1] and not s.l[i + 1] then s.l[i + 1] = TRUE end
    end
    i = i + 1
  end
end
guide_vault_open = function(r) return r >= 252 and r <= 286 end
-- phase k (4 down to 1): turn d until tumbler k is engaged and open while k+1.. stay where they are
guide_vault_plan = function(s, k, d, plan)
  if k == 0 then return plan end
  local c = guide_vault_clone(s)
  if k < 4 then guide_vault_reverse(c) end
  local n = 0
  local kept = { }
  local i = k + 1
  while i <= 4 do kept[i] = c.r[i] i = i + 1 end
  while n < 1500 do
    guide_vault_step(c, d)
    n = n + 1
    i = k + 1
    while i <= 4 do
      if c.r[i] ~= kept[i] then return nil end
      i = i + 1
    end
    if c.l[k] and guide_vault_open(c.r[k]) then
      local rest = { }
      local j = 1
      while plan[j] do rest[j] = plan[j] j = j + 1 end
      rest[j] = { d = d, n = n }
      local found = guide_vault_plan(c, k - 1, -d, rest)
      if found then return found end
      -- else try a few degrees further, while k stays open
    end
  end
  return nil
end
guide_vault_turn = function(d, n)
  local sd = vd.safe_door
  if d == 1 then sd:spin_right(0) else sd:spin_left(0) end
  break_here()
  break_here()
  manny:play_chore_looping(49, manny.base_costume)
  local done = 0
  while done < n do
    local k = 6
    if n - done < k then k = n - done end
    if d == 1 then sd:spin_right(k) else sd:spin_left(k) end
    local j = 0
    while j < k do
      if d == 1 then sd:spin_tumblers_right() else sd:spin_tumblers_left() end
      j = j + 1
    end
    vd.play_wheel_sfx()
    done = done + k
    break_here()
  end
  manny:set_chore_looping(49, FALSE, manny.base_costume)
  manny:wait_for_chore(49, manny.base_costume)
  sleep_for(400)
end
guide_vault = function()
  local plan = guide_vault_plan(guide_vault_copy(vd.safe_door.tumblers), 4, 1, { })
  if not plan then
    guide_error = "no plan for the vault's tumblers"
    return
  end
  local i = 1
  while plan[i] do
    guide_vault_turn(plan[i].d, plan[i].n)
    i = i + 1
  end
end
-- the broadaxe (vo, vi): turned and dragged backwards as the keys do (vo.turn_left, vo.walk_backward)
guide_axe_face = function(x, y, off)
  local want = GetActorYawToPoint(manny.hActor, { x = x, y = y, z = 0 }) + (off or 0)
  manny:play_chore_looping(mn_lift_ax_drag_ax, "mn_lift_ax.cos")
  start_sfx("voAxTDrg.imu")
  local n = 0
  while n < 60 do
    local r = manny:getrot()
    local d = want - r.y
    while d > 180 do d = d - 360 end
    while d < -180 do d = d + 360 end
    if abs(d) <= 10 then
      manny:setrot(r.x, want, r.z, TRUE)
      n = 60
    elseif d > 0 then
      manny:setrot(r.x, r.y + 10, r.z, TRUE)
    else
      manny:setrot(r.x, r.y - 10, r.z, TRUE)
    end
    break_here()
    n = n + 1
  end
  stop_sound("voAxTDrg.imu")
  manny:stop_chore(mn_lift_ax_drag_ax, "mn_lift_ax.cos")
  manny:play_chore_looping(mn_lift_ax_hold_ax, "mn_lift_ax.cos")
end

    break_here()
    n = n + 1
  end

  manny:play_chore_looping(mn_lift_ax_hold_ax, "mn_lift_ax.cos")
end
guide_axe_dist = function()
  local q = manny:getpos()
  return sqrt((q.x + 0.236278996) ^ 2 + (q.y + 0.387933999) ^ 2)
end
guide_axe_near_tile = function() return guide_axe_dist() <= 0.3 end
guide_axe_off_tile = function() return guide_axe_dist() >= 0.25 end
guide_axe_in_vi = function() return system.currentSet == vi end
-- the axe from the vault, backwards through the secret door, onto the tile the water runs through (drop_test)
guide_axe = function()
  guide_sentence("use", vo.broadaxe)
  local p = manny:getpos()
  guide_axe_face(p.x + (p.x + 1.346), p.y + (p.y - 1.032))
  guide_axe_back(guide_axe_in_vi)
  while cutSceneLevel > 0 or vo.manny_exiting do break_here() end
  -- the game's drop_test: 0.15 to 0.43 from the tile, facing it within 30 degrees, but not exactly (the angle
  -- of two equal vectors comes out NaN there); backwards only, so too close is mended by facing the tile and
  -- backing off
  local n = 0
  while n < 4 and not drop_test() do
    p = manny:getpos()
    if guide_axe_dist() > 0.35 then
      guide_axe_face(p.x + (p.x + 0.236278996), p.y + (p.y + 0.387933999))
      guide_axe_back(guide_axe_near_tile)
    elseif guide_axe_dist() < 0.2 then
      guide_axe_face(-0.236278996, -0.387933999)
      guide_axe_back(guide_axe_off_tile)
    end
    guide_axe_face(-0.236278996, -0.387933999, 8)
    n = n + 1
  end
  guide_wait_for(start_script(vo.drop_axe))
end
